# 本工程仅构建弹体审阅工具与碰撞测试，不链接游戏引擎。
cmake_minimum_required(VERSION 3.21)
project(projectile_review LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REVIEW_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
set(CUSTOMIZE_BUILD ON CACHE BOOL "" FORCE)
set(SUPPORT_FILEFORMAT_GLTF ON CACHE BOOL "" FORCE)
add_subdirectory("${REVIEW_ROOT}/.deps/raylib-5.5" "${CMAKE_BINARY_DIR}/raylib")
add_library(projectile_collision STATIC collision.cpp)
target_include_directories(projectile_collision PUBLIC .)
target_link_libraries(projectile_collision PUBLIC raylib)
add_library(projectile_classic STATIC classic_renderer.cpp)
add_library(projectile_collection STATIC collection_renderer.cpp)
target_link_libraries(projectile_collection PUBLIC projectile_classic)
add_executable(collection_inspector collection_inspector.cpp)
target_link_libraries(collection_inspector PRIVATE projectile_collection)
add_executable(collection_tests collection_tests.cpp)
target_link_libraries(collection_tests PRIVATE projectile_collection)
add_executable(collection_stress collection_stress.cpp)
target_link_libraries(collection_stress PRIVATE projectile_collection)
target_include_directories(collection_stress PRIVATE "${REVIEW_ROOT}/src")
target_link_libraries(projectile_classic PUBLIC projectile_collision)
add_executable(classic_inspector classic_inspector.cpp)
target_link_libraries(classic_inspector PRIVATE projectile_classic)
add_executable(projectile_stress stress.cpp)
target_link_libraries(projectile_stress PRIVATE projectile_classic)
target_include_directories(projectile_stress PRIVATE "${REVIEW_ROOT}/src")
add_executable(classic_tests classic_tests.cpp)
target_link_libraries(classic_tests PRIVATE projectile_classic)
if(MINGW)
  target_link_options(collection_inspector PRIVATE -static-libgcc -static-libstdc++ -static)
  target_link_options(collection_tests PRIVATE -static-libgcc -static-libstdc++ -static)
  target_link_options(collection_stress PRIVATE -static-libgcc -static-libstdc++ -static)
  target_link_options(classic_inspector PRIVATE -static-libgcc -static-libstdc++ -static)
  target_link_options(projectile_stress PRIVATE -static-libgcc -static-libstdc++ -static)
  target_link_options(classic_tests PRIVATE -static-libgcc -static-libstdc++ -static)
endif()
enable_testing()
add_library(projectile_laser STATIC laser_renderer.cpp)
target_link_libraries(projectile_laser PUBLIC projectile_classic)
foreach(name laser_inspector laser_tests laser_stress)
  add_executable(${name} ${name}.cpp)
  target_link_libraries(${name} PRIVATE projectile_laser projectile_collection)
  target_include_directories(${name} PRIVATE "${REVIEW_ROOT}/src")
  if(MINGW)
    target_link_options(${name} PRIVATE -static-libgcc -static-libstdc++ -static)
  endif()
endforeach()
add_test(NAME laser_geometry_and_timing COMMAND laser_tests "${REVIEW_ROOT}")
add_test(NAME collection_all_types COMMAND collection_tests "${REVIEW_ROOT}")
add_test(NAME classic_geometry_and_renderer COMMAND classic_tests "${REVIEW_ROOT}")

add_executable(reimu_inspector reimu_inspector.cpp)
target_link_libraries(reimu_inspector PRIVATE projectile_collision)
if(MINGW)
 target_link_options(reimu_inspector PRIVATE -static-libgcc -static-libstdc++ -static)
endif()

add_executable(marisa_inspector marisa_inspector.cpp)
target_link_libraries(marisa_inspector PRIVATE projectile_collision)
if(MINGW)
 target_link_options(marisa_inspector PRIVATE -static-libgcc -static-libstdc++ -static)
endif()
add_executable(marisa_tests marisa_tests.cpp)
target_link_libraries(marisa_tests PRIVATE projectile_collision)
if(MINGW)
 target_link_options(marisa_tests PRIVATE -static-libgcc -static-libstdc++ -static)
endif()
add_test(NAME marisa_geometry COMMAND marisa_tests "${REVIEW_ROOT}")

add_library(bomb_review_volume STATIC bomb_review_math.cpp "${REVIEW_ROOT}/src/gameplay/effects/bomb_volume.cpp")
target_include_directories(bomb_review_volume PUBLIC "${REVIEW_ROOT}/src")
target_link_libraries(bomb_review_volume PUBLIC raylib)
add_executable(bomb_inspector bomb_inspector.cpp)
target_link_libraries(bomb_inspector PRIVATE projectile_collision bomb_review_volume)
if(MINGW)
 target_link_options(bomb_inspector PRIVATE -static-libgcc -static-libstdc++ -static)
endif()
add_executable(bomb_tests bomb_tests.cpp)
target_link_libraries(bomb_tests PRIVATE projectile_collision bomb_review_volume)
if(MINGW)
 target_link_options(bomb_tests PRIVATE -static-libgcc -static-libstdc++ -static)
endif()
add_test(NAME bomb_geometry COMMAND bomb_tests "${REVIEW_ROOT}")
